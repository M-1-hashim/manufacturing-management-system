    90→// درخواست JSON با پیام خطای دری
    91→async function jsonReq(url: string, method: string, body?: unknown) {
    92→  const res = await fetch(url, {
    93→    method,
    94→    headers: body ? { 'Content-Type': 'application/json' } : undefined,
    95→    body: body ? JSON.stringify(body) : undefined,
    96→  })
    97→  const json = await res.json().catch(() => null)
    98→  if (!res.ok) throw new Error(json?.error || `خطا در عملیات (${res.status})`)
    99→  return json
   100→}

   103→function toISODate(d: Date) {
   104→  const pad = (x: number) => String(x).padStart(2, '0')
   105→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
   106→}
