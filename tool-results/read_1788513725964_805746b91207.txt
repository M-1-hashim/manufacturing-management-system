    91→    91→async function jsonReq(url: string, method: string, body?: unknown) {
    92→    92→  const res = await fetch(url, {
    93→    93→    method,
    94→    94→    headers: body ? { 'Content-Type': 'application/json' } : undefined,
    95→    95→    body: body ? JSON.stringify(body) : undefined,
    96→    96→  })
    97→    97→  const json = await res.json().catch(() => null)
    98→    98→  if (!res.ok) throw new Error(json?.error || `خطا در عملیات (${res.status})`)
    99→    99→  return json
   100→   100→}
